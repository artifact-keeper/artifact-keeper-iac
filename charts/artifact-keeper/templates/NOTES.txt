{{- if .Values.ingress.enabled }}
  URL: {{ if .Values.ingress.tls.enabled }}https{{ else }}http{{ end }}://{{ .Values.ingress.host }}
{{- end }}

{{- if .Values.httpRoute.enabled }}
HTTPRoute: {{ include "artifact-keeper.fullname" . }}
  Hostnames: {{ join ", " .Values.httpRoute.hostnames }}
  The referenced Gateway listener controls HTTP/HTTPS and TLS certificates.
  Check the route's Accepted and ResolvedRefs status and the Gateway's
  allowedRoutes before sending traffic.
  {{- if .Values.httpRoute.dtrack.enabled }}
  /dtrack is publicly routed without rewriting, as with Ingress. Protect it
  with authentication you control.
  {{- end }}
{{- end }}

Backend API:
  kubectl port-forward svc/{{ include "artifact-keeper.fullname" . }}-backend {{ .Values.backend.service.httpPort }}:{{ .Values.backend.service.httpPort }}

Web UI:
  kubectl port-forward svc/{{ include "artifact-keeper.fullname" . }}-web {{ .Values.web.service.port }}:{{ .Values.web.service.port }}

{{- if .Values.postgres.enabled }}

PostgreSQL (in-cluster):
  kubectl port-forward svc/{{ include "artifact-keeper.fullname" . }}-postgres 5432:5432
{{- end }}

{{- if .Values.dependencyTrack.enabled }}

DependencyTrack:
  kubectl port-forward svc/{{ include "artifact-keeper.fullname" . }}-dtrack 8092:8080
  {{- if .Values.ingress.enabled }}
  {{- if .Values.ingress.dtrack.enabled }}
  The Dependency-Track UI/API is also routed publicly at /dtrack on the
  ingress host (ingress.dtrack.enabled=true). Make sure it sits behind
  authentication you control.
  {{- else }}
  The Dependency-Track UI/API is NOT exposed on the ingress. Use the
  port-forward above, or set ingress.dtrack.enabled=true to route /dtrack
  publicly.
  {{- end }}
  {{- end }}
{{- end }}

{{- if eq (include "artifact-keeper.allowHttpIntegrations" .) "true" }}

WARNING: ALLOW_HTTP_INTEGRATIONS=1 is set on the backend
  (backend.allowHttpIntegrations={{ .Values.backend.allowHttpIntegrations | default "auto" }}).
  The backend may call integrations over plain HTTP. This relaxes outbound
  TLS posture for ALL integrations, not just the bundled Dependency-Track.
  Set backend.allowHttpIntegrations: "false" (or ALLOW_HTTP_INTEGRATIONS in
  backend.env) to turn it off; note the bundled Dependency-Track
  integration requires it unless you put TLS in front of Dependency-Track.
{{- end }}

{{- $benv := .Values.backend.env | default dict }}
{{- $presign := has (lower (toString (get $benv "PRESIGNED_DOWNLOADS_ENABLED"))) (list "true" "1") }}
{{- $s3Redirect := has (lower (toString (get $benv "S3_REDIRECT_DOWNLOADS"))) (list "true" "1") }}
{{- $azRedirect := has (lower (toString (get $benv "AZURE_REDIRECT_DOWNLOADS"))) (list "true" "1") }}
{{- if and $presign $s3Redirect (get $benv "S3_ENDPOINT") (not (get $benv "S3_PUBLIC_ENDPOINT")) }}

WARNING: presigned S3 downloads are on but backend.env.S3_PUBLIC_ENDPOINT
  is unset. Redirects (including the OCI blob 307) point clients at
  S3_ENDPOINT={{ get $benv "S3_ENDPOINT" }}; if clients cannot reach that
  address, every redirected download fails. Set S3_PUBLIC_ENDPOINT to the
  scheme://host[:port] clients use, forwarded to the object store with the
  Host header and path unchanged (SigV4 signs both).
{{- end }}
{{- if and $presign $azRedirect (get $benv "AZURE_STORAGE_ENDPOINT") (not (get $benv "AZURE_STORAGE_PUBLIC_ENDPOINT")) }}

WARNING: presigned Azure downloads are on but
  backend.env.AZURE_STORAGE_PUBLIC_ENDPOINT is unset. SAS redirects point
  clients at AZURE_STORAGE_ENDPOINT={{ get $benv "AZURE_STORAGE_ENDPOINT" }};
  set AZURE_STORAGE_PUBLIC_ENDPOINT if clients cannot reach it.
{{- end }}

{{- if .Values.serviceMonitor.enabled }}

Monitoring:
  ServiceMonitor is enabled. Prometheus will scrape /metrics from the backend.
{{- end }}

{{- if eq (include "artifact-keeper.fleet.enabled" .) "true" }}

CAUTION: Fleet mode shares ONE object-storage bucket across instances,
  separated only by key prefix. Isolation holds ONLY if each instance uses
  its own credentials (fleet.storage.existingSecret) scoped to its own
  prefix. Reusing one Secret across instances lets every instance read and
  overwrite every other instance's artifacts.
{{- end }}
